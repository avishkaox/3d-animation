import { useRef, useEffect, useMemo } from 'react';
import { useFrame, useThree } from '@react-three/fiber';
import { gsap } from 'gsap';
import * as THREE from 'three';
import { EffectComposer, Bloom } from '@react-three/postprocessing';

function Signal({ curve, targetPoint, initialT = 0 }) {
  const meshRef = useRef();
  const tRef = useRef(initialT);
  const dir = useRef(1);
  const tweenRef = useRef();

  const sampled = useMemo(() =>
    curve ? Array.from({ length: 101 }, (_, i) => {
      const t = i / 100;
      return { t, point: curve.getPoint(t) };
    }) : [],
    [curve]
  );

  useFrame(() => {
    if (meshRef.current && !targetPoint) {
      tRef.current += 0.001 * dir.current;
      if (tRef.current >= 1 || tRef.current <= 0) dir.current *= -1;
      meshRef.current.position.copy(curve.getPoint(tRef.current));
    }
  });

  useEffect(() => {
    if (!targetPoint || !curve) return;
    const { t: closestT } = sampled.reduce((acc, cur) => {
      const d = cur.point.distanceTo(targetPoint);
      return d < acc.d ? { t: cur.t, d } : acc;
    }, { t: 0, d: Infinity });
    tweenRef.current?.kill();
    tweenRef.current = gsap.to(tRef, {
      current: closestT,
      duration: Math.abs(closestT - tRef.current) * 5,
      ease: 'back.out(1.7)',
      onUpdate: () => {
        const t = THREE.MathUtils.clamp(tRef.current, 0, 1);
        meshRef.current?.position.copy(curve.getPoint(t));
      },
    });
    return () => tweenRef.current?.kill();
  }, [targetPoint, curve, sampled]);

  return (
    <mesh ref={meshRef}>
      <sphereGeometry args={[0.2, 16, 16]} />
      <meshStandardMaterial
        color="yellow"
        emissive="yellow"
        emissiveIntensity={5}
        toneMapped={false}
        metalness={0.3}
        roughness={0.4}
      />
    </mesh>
  );
}

export default function SceneWithBloom(props) {
  return (
    <>
      {/** Place lights + other objects */}
      <Signal {...props} />
      <EffectComposer disableNormalPass>
        <Bloom
          luminanceThreshold={1}
          luminanceSmoothing={0.025}
          intensity={1}
          mipmapBlur
        />
      </EffectComposer>
    </>
  );
}
